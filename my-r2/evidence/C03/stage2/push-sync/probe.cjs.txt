// Push-sync probe (evidence): wraps the per-frame tween writer and reads back the COMPUTED transform / opacity of every moving layer
// right after each write. node probe3.cjs <jump> <openSel> <surface> [reverseAfterMs]
const L = require('/home/user/JU/my-r2/tools/_lib.cjs');
(async () => {
  const [jump, sel, surf, rev] = process.argv.slice(2);
  const { b, p } = await L.launch({ viewport: { width: 480, height: 1000 } });
  await L.openConcept(p, '/home/user/JU/.build/C03.html', 'C03', 1200);
  await L.E(p, 'C03', "var S = JU.shared && JU.shared.skins; if (S && inst.envRef && inst.envRef.current) S.pause(inst.envRef.current);");
  await L.jump(p, 'C03', jump, 900); await p.waitForTimeout(400);
  await L.E(p, 'C03', `window.__pl = []; var orig = inst._twApply;
    var tx = function (el) { if (!el) return null; var m = getComputedStyle(el).transform; if (!m || m === 'none') return 0; var v = m.match(/matrix\\(([^)]+)\\)/); return v ? Math.round(parseFloat(v[1].split(',')[4]) * 10) / 10 : m; };
    var op = function (el) { return el ? Math.round(parseFloat(getComputedStyle(el).opacity) * 1000) / 1000 : null; };
    inst._twApply = function (x, now) { orig.call(this, x, now); var tw = this._tw; if (!tw) return; var home = this.homeRef.current;
      var pg = tw.tops.filter(function (e) { return !e.classList.contains('is-leaving'); })[0] || tw.tops[0], un = tw.unders[0];
      window.__pl.push({ kind: tw.fwd ? 'push' : tw.swap ? 'replace' : 'pop', at: Math.round((now || performance.now()) * 10) / 10, x: Math.round(x * 1000) / 1000,
        newPage: pg ? (pg.getAttribute('data-surface') + ' x=' + tx(pg)) : null, under: un ? ((un.getAttribute('data-surface') || 'home face') + ' x=' + tx(un) + ' o=' + op(un)) : null,
        lift: home ? getComputedStyle(home).getPropertyValue('--c03-dp').trim() : '', peek: home ? getComputedStyle(home).getPropertyValue(tw.pk).trim() : '' }); };`);
  const t0 = Date.now();
  await L.clickSel(p, 'C03', sel, {});
  if (rev) { await p.waitForTimeout(+rev); await L.clickSel(p, 'C03', '[data-surface="' + surf + '"] [data-act="back"]', {}); await p.waitForTimeout(900); }
  else { await p.waitForTimeout(900); await L.clickSel(p, 'C03', '[data-surface="' + surf + '"] [data-act="back"]', {}); await p.waitForTimeout(900); }
  const rows = await p.evaluate(() => window.__pl);
  const base = rows.length ? rows[0].at : 0; rows.forEach(r => { r.t = Math.round((r.at - base) * 10) / 10; delete r.at; });
  console.log(JSON.stringify(rows));
  await b.close();
})();
