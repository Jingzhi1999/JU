// positional double / triple tap on the first draft's 删除 (real mouse at a fixed point), RM on/off
const { chromium } = require('/opt/node-tools/node_modules/playwright');
const path = require('path'); const rm = process.argv[2] === 'rm'; const gap = +(process.argv[3] || 80);
(async () => {
  const b = await chromium.launch(); const p = await b.newPage({ viewport: { width: 1000, height: 1000 } });
  const errors = []; p.on('pageerror', e => errors.push(e.message));
  await p.goto('file://' + path.resolve(process.env.JU_STANDALONE) + '#concept/C09'); await p.waitForTimeout(1200);
  const M = "window.JUShell.mounts.stage['stage:C09:']"; const ev = (js) => p.evaluate(js);
  await ev(`${M}.instance.jumpTo('h-drafts-many')`); await p.waitForTimeout(1000);
  if (rm) await ev(`${M}.instance.toggleRM()`); await p.waitForTimeout(300);
  const titles = () => ev(`[].map.call(${M}.root.querySelectorAll('.c09-feed .c09-draft .c09-cr-ttl'),function(e){return e.textContent.trim();})`);
  const before = await titles();
  const pt = await ev(`(function(){var r=${M}.root.querySelector('.c09-feed .c09-draft [data-act="delete"]').getBoundingClientRect();return [r.left+r.width/2,r.top+r.height/2];})()`);
  for (let i = 0; i < 3; i++) { await p.mouse.click(pt[0], pt[1]); await p.waitForTimeout(gap); }
  await p.waitForTimeout(600);
  const after = await titles(); const engine = await ev(`(${M}.instance.state.data.drafts||[]).map(function(d){return d.title;})`);
  const bar = await ev(`((${M}.root.querySelector('.c09-undo')||{}).innerText||'').replace(/\\s+/g,' ')`);
  const pink = await ev(`!!${M}.root.querySelector('.c09-feed .c09-ialert, .c09-err')`);
  const pt2 = await ev(`(function(){var e=${M}.root.querySelector('.c09-feed .c09-draft [data-act="delete"]');if(!e)return null;var r=e.getBoundingClientRect();return [Math.round(r.left+r.width/2),Math.round(r.top+r.height/2)];})()`);
  console.log(JSON.stringify({ rm, gap, pt: pt.map(Math.round), nextDeleteAt: pt2, before, after, engine, bar, pink, errors })); await b.close();
})();
